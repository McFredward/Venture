cmake_minimum_required(VERSION 3.28)
project(VentureGuestTests LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Configure with the NDK android.toolchain.cmake (see build.py)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
option(VTEST_SHARED_ONLY "Build the cross-DSO C++ test with libc++_shared" OFF)
add_compile_options(-O2 -g -fexceptions -frtti -fno-builtin -Wall -Wextra -Werror)
add_compile_definitions(_GNU_SOURCE)
# Old Quest titles use emutls; current Unity builds use native ELF TLS.
if(ANDROID_PLATFORM_LEVEL LESS 29)
  add_compile_options(-femulated-tls)
  add_link_options("-Wl,--pack-dyn-relocs=android")
else()
  add_compile_options(-fno-emulated-tls)
  add_link_options("-Wl,--pack-dyn-relocs=android+relr" "-Wl,--no-use-android-relr-tags")
endif()
if(ANDROID_ABI STREQUAL "arm64-v8a")
  add_compile_options(-moutline-atomics)
else()
  add_compile_options(-mfpu=neon-fp16)
endif()
add_link_options("-Wl,--no-undefined" "-Wl,--build-id=sha1")

function(vtest_test name)
  add_library(vtest_${name} SHARED src/${name}.cpp)
  target_include_directories(vtest_${name} PRIVATE include)
  target_link_libraries(vtest_${name} PRIVATE dl m atomic)
endfunction()
if(VTEST_SHARED_ONLY)
  add_library(vtest_cpp_peer SHARED src/cpp_peer.cpp)
  target_include_directories(vtest_cpp_peer PRIVATE include)
  add_library(vtest_cpp_shared SHARED src/cpp.cpp)
  target_include_directories(vtest_cpp_shared PRIVATE include)
  target_compile_definitions(vtest_cpp_shared PRIVATE VTEST_CPP_NAME="cpp_shared")
  target_link_libraries(vtest_cpp_shared PRIVATE vtest_cpp_peer dl)
else()
  foreach(name strings printf parse malloc libm cpp lifecycle pthreads tls atomics
      jumps signals time files dl misc log neon jni ndk_api)
    vtest_test(${name})
  endforeach()
  foreach(peer cpp_peer lifecycle_peer tls_peer dl_peer)
    add_library(vtest_${peer} SHARED src/${peer}.cpp)
    target_include_directories(vtest_${peer} PRIVATE include)
  endforeach()
  target_link_libraries(vtest_cpp PRIVATE vtest_cpp_peer)
  target_link_libraries(vtest_log PRIVATE log)
  target_link_libraries(vtest_ndk_api PRIVATE android)
  target_link_libraries(vtest_lifecycle_peer PRIVATE dl)
  add_executable(vtest_runner src/runner.cpp)
  target_link_libraries(vtest_runner PRIVATE dl)
  target_compile_options(vtest_runner PRIVATE -fPIE)
  target_link_options(vtest_runner PRIVATE -pie)
endif()
